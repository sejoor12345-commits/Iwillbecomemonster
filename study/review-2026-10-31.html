const EXERCISES = {

function lastTime(split, name) {                     // 같은 분할(조합 그대로)의 지난번 무게

5-2. B1 — 다른 기기에서 고친 식사가 이 폰에 안 바뀜


비유: 칠판(Supabase) 하나, 공책 두 권(폰 A · 폰 B). 폰 B는 "칠판에 있는데 내 공책에 없는 줄만 베껴 적기"를 했어. 그래서 폰 A가 칠판의 420을 600으로 고쳐도, 폰 B 공책에는 이미 420줄이 있으니까 그냥 건너뜀 → 영원히 420. 에러는 하나도 없어.


고친 규칙: 내 공책의 줄이 이미 칠판에 옮겨 적었던 줄(synced: true)이면 칠판이 기준 → 칠판 값으로 바꿔 적기. 내가 고치고 아직 안 옮긴 줄(synced: false)과 방금 옮긴 줄(justUploaded)은 내 공책이 최신이니까 그대로.




  
실행 셀 5-1

  
const local  = [{ id: "m1", kcal: 420, synced: true }, { id: "m2", kcal: 500, synced: false }];
const remote = [{ id: "m1", kcal: 600 }, { id: "m2", kcal: 300 }, { id: "m3", kcal: 250 }];
const show = list => list.map(m => m.id + ":" + m.kcal).join(" ");

function oldDownload() {                     // v0.6-c까지: 폰에 있으면 건너뛰기
  const out = local.map(m => ({ ...m }));
  const ids = new Set(out.map(m => m.id));
  for (const r of remote) {
    if (ids.has(r.id)) continue;
    out.push({ ...r, synced: true });
  }
  return show(out);
}
function newDownload() {                     // v0.6-d: 올라가 있던 것(synced)은 인터넷 값으로
  const out = local.map(m => ({ ...m }));
  for (const r of remote) {
    const i = out.findIndex(m => m.id === r.id);
    if (i < 0) out.push({ ...r, synced: true });
    else if (out[i].synced) out[i] = { ...r, synced: true };
  }
  return show(out);
}
console.log(oldDownload());   // ①
console.log(newDownload());   // ②
