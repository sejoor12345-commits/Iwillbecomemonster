3-5. escapeHtml — 사용자가 적은 글자는 꼭 바꿔서 넣기




  
실행 셀 3-2

  
function escapeHtml(text) {
  return String(text).replace(/[&<>"']/g, c => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]));
}
const food = "<i>닭</i> 가슴살";
console.log(`<b>${food}</b>`);               // ① 그대로 넣으면 (화면에서는 '닭'이 기울어짐)
console.log(`<b>${escapeHtml(food)}</b>`);   // ② 바꿔서 넣으면 (화면에 글자 그대로 보임)

  




음식 이름, 종목 이름처럼 사용자가 적은 글자는 화면에 넣기 전에 escapeHtml()을 거쳐. 안 그러면 글자가 HTML로 해석돼서 모양이 깨지거나, 나쁜 코드가 들어갈 수도 있어.

function nextWeight(weight, reps, low, high, step, rpe) {
  if (weight === 0) return { weight: 0, reason: "맨몸 → 1회씩 더" };
  if (Math.min(...reps) >= high && rpe === 10) return { weight, reason: "굳히기" };
  if (Math.min(...reps) >= high) return { weight: weight + step, reason: "증량" };
  if (Math.max(...reps) < low) return { weight: Math.max(weight - step, 0), reason: "감량" };
  return { weight, reason: "같은 무게" };
}
